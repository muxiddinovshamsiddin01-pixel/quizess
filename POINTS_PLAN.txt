━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Баллы считаются ТОЛЬКО В КОНЦЕ квиза (функция showDone в quiz.js)
   → submitQuizResult({ subject, mode, score, total, pct, time_seconds })

2. calc_points в main.py считает баллы по формуле:
   base = pct * total * 0.5
   bonus = 50 (100%) / 25 (≥90%) / 10 (≥75%) / 0
   speed = max(0, 30 - time_seconds//10)
   participation = +10
   → итого: всё одним числом после завершения

3. Нет никакой обратной связи во время квиза — ни "+10 pts" flash,
   ни отображения накопленных баллов в прогресс-баре

4. Режим "Mistakes only" запускается через:
   retryMistakes() → url?mode=mistakes&seed=...
   mistakeIds сохраняются в localStorage через S.set('mistakes', ...)

5. Режим mistakes над mistakes (работа над ошибками ошибок):
   После первого mistakes-квиза те ID что снова неправильные
   → pushState → попадают опять в S.set('mistakes', ...) → можно
   снова запустить mistakes режим. Это уже работает рекурсивно!
   Т.е. "работа над ошибками ошибок" технически уже реализована —
   просто кнопка "Retry Mistakes" доступна в конце любого квиза.

ФОРМУЛА calcLivePoints (простая, понятная):
  Правильный ответ → +10 pts базово
  + бонус за скорость: если ответил быстро (< 10 сек на вопрос) → +5
  Неправильный → 0 pts
  (финальный бэкенд-расчёт остаётся как есть для суммарного зачисления)

  function showPointsFlash(pts, isOk) {
    const existing = document.getElementById('ptsFlash');
    if (existing) existing.remove();

    const el = document.createElement('div');
    el.id = 'ptsFlash';
    el.style.cssText = `
      position: fixed;
      top: 70px;
      right: 20px;
      z-index: 9999;
      font-size: 22px;
      font-weight: 800;
      color: ${isOk ? 'var(--gr)' : 'var(--re)'};
      opacity: 1;
      transition: opacity 0.8s ease, transform 0.8s ease;
      pointer-events: none;
      font-family: 'Space Mono', monospace;
    `;
    el.textContent = isOk ? `+${pts} ⭐` : '−0';
    document.body.appendChild(el);

    requestAnimationFrame(() => {
      setTimeout(() => {
        el.style.opacity = '0';
        el.style.transform = 'translateY(-20px)';
        setTimeout(() => el.remove(), 800);
      }, 700);
    });
  }
